Row level security

Your posting says the dangerous write should be impossible, not discouraged. I agree, so here's that rule as a policy you can try to break.

/* Agents may read, report, and carry a human's update
   once a human consents. They never write a commitment. */
alter table work_items enable row level security;

create policy "agents_read" on work_items
  for select to agent using (true);

create policy "agents_carry_consented_updates"
  on work_items for update to agent
  using (human_consented(id))
  with check (
    owner     is not distinct from old_owner(id)
    and due_date  is not distinct from old_due(id)
    and status    is not distinct from old_status(id)
  );

/* and the credential itself has no grant on those columns */
revoke update (owner, due_date, status)
  on work_items from agent;
